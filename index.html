async function checkGPS(){ const t = i18n[currentLang]; if(isFirstCheck) document.getElementById('status').innerText = t.verifying; navigator.geolocation.getCurrentPosition(async (pos) => { const lat=pos.coords.latitude, lon=pos.coords.longitude, acc=Math.round(pos.coords.accuracy); const city = await getCityCountry(lat, lon); // === SAMA FINAL - LAT/LON BASED - NO LANGUAGE BUG === // Saudi bounding box: Lat 16-32, Lon 34-56 const isInSaudi = (lat >= 16.3 && lat <= 32.1 && lon >= 34.5 && lon <= 55.6); let isMock=false, reason="", risk=100; if(acc <= 2){ // 0-2m = Hamesha Mock isMock=true; risk=960; reason=`Perfect ${acc}m = 100% Mock`; } else if(!isInSaudi && acc <= 15){ // Saudi ke bahar (Belabo) + accuracy achi = Mock isMock=true; risk=960; reason=`Foreign Location ${city} - ${acc}m = Mock`; } else { // Saudi ke andar 3m-100m sab Clean isMock=false; risk=100; reason=`Clean - ${city} - ${acc}m`; } lastData={lat, lon, acc, city, isMock, reason, risk, isInSaudi}; renderUI(lastData); isFirstCheck=false; }, ()=>{ document.getElementById('status').innerText="Location permission required"; isFirstCheck=false; }, {enableHighAccuracy:true, maximumAge:0, timeout:10000}); } function setLang(lang){ currentLang = lang; localStorage.setItem('sama_lang', lang); document.documentElement.lang = lang; document.documentElement.dir = lang==='ar'?'rtl':'ltr'; document.getElementById('btnEN').className = lang==='en'?'active-lang':'inactive-lang'; document.getElementById('btnAR').className = lang==='ar'?'active-lang':'inactive-lang'; // Language badalte hi naya city fetch karo - taaki AR me bhi sahi city aaye if(lastData){ getCityCountry(lastData.lat, lastData.lon).then(newCity=>{ lastData.city = newCity; renderUI(lastData); }); } else { checkGPS(); } }
